cmake_minimum_required(VERSION 3.20)
project(SoundConnectKDE LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# The presentation is a separate process, so Plasma 5 can use its own native
# Qt 5/KF5 widgets while the shared device backend retains Qt 6.5+.
find_package(KF6WidgetsAddons QUIET)
if(KF6WidgetsAddons_FOUND)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    find_package(KF6StatusNotifierItem REQUIRED)
    find_package(KF6ColorScheme REQUIRED)
    set(native_libraries Qt6::Widgets KF6::WidgetsAddons KF6::StatusNotifierItem KF6::ColorScheme)
else()
    find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets)
    find_package(KF5WidgetsAddons REQUIRED)
    find_package(KF5Notifications REQUIRED)
    find_package(KF5ConfigWidgets 5.89 REQUIRED)
    set(native_libraries Qt5::Widgets KF5::WidgetsAddons KF5::Notifications KF5::ConfigWidgets)
endif()
add_executable(SoundConnectKDE main.cpp)
target_link_libraries(SoundConnectKDE PRIVATE ${native_libraries})
configure_file(../../clients/common/catalog.json catalog.json COPYONLY)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(kde_transport_tests TransportTests.cpp)
    target_link_libraries(kde_transport_tests PRIVATE ${native_libraries})
    add_test(NAME kde-transport COMMAND kde_transport_tests ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/../../clients/common/test_process_client.py")
endif()
